# SAMPLE LOAD (U8a): SAMPLE LOAD opens the kit's file picker on Apps/o3_voc03er/ (.wav only); UP and DOWN name the slot; F1 loads the file under the cursor into it;
# F8 returns to the screen it came from. A mono file, a stereo file over the same slot (REPLACES SLOT 1, F1 confirms: same length, so the slot is cleared first), a 48 kHz
# 24-bit stereo file converted on load to the slot's 44.1 kHz 16-bit (the strip says so), any other key cancelling the ask, a damaged file and a 192 kHz file refused, a 20 s
# 48 kHz 24-bit file converted and clipped to 16 s (the read stops when the stage is full), a slot that records refused. The simulator shows the screen and the unit's length; it cannot say how it sounds. The card needs the WAVs of sample_load_wavs.py:
#   mkdir -p <card> <dir> && python3 -I app/scenarios/sample_load_wavs.py <card>/Apps/o3_voc03er
#   build-host/app/o3_voc03er_sim --scenario app/scenarios/sample_load.txt --card <card> --out <dir>
# Files sorted: a_mono, b_stereo (same length as a_mono), c_48k24 (the same length once converted), d_8bit, e_long20, f_broken, g_192k.
wait 300
expect state card=ready
tap sampleload
wait 600
snap 00_list
expect state loop1=0

# a_mono.wav (1.5 s) into the empty slot 1: no ask
tap f1
wait 150
snap 01_reading
wait 600
snap 02_loaded
expect state screen=SAMPLE_LOAD

# b_stereo.wav into the same slot: it holds a sample, so it asks; F1 confirms; the slot keeps the length
turn 1
tap f1
wait 100
snap 03_ask
tap f1
wait 1200
snap 04_replaced
expect state screen=SAMPLE_LOAD

# c_48k24.wav (1.5 s, 48 kHz, 24-bit stereo): asks, F1 confirms, then the file is converted to 44.1 kHz 16-bit mono, 66150 frames, and the strip says what was done (the list
# starts at its top after every job)
turn 2
tap f1
wait 100
tap f1
snap 05a_converting
wait 2500
snap 05_converted
expect state screen=SAMPLE_LOAD
expect state loop1=66150

# d_8bit.wav: asks; any other key cancels (F3 here); an 8-bit file loads, but not this time
turn 3
tap f1
wait 100
tap f3
wait 300
snap 06_not_replaced
expect state screen=SAMPLE_LOAD

# f_broken.wav: the data chunk is longer than the file
turn 5
tap f1
wait 100
tap f1
wait 800
snap 07_refused_broken
expect state screen=SAMPLE_LOAD

# g_192k.wav: refused at its header, the slot untouched
turn 6
tap f1
wait 100
tap f1
wait 800
snap 07b_refused_rate
expect state screen=SAMPLE_LOAD
expect state loop1=66150

# e_long20.wav (20 s of stereo) into the empty slot 2: the first 16 s; the strip says so
tap down
wait 100
snap 08_slot2
turn 4
tap f1
wait 40
snap 09_reading_long
wait 1500
snap 10_clipped
wait 8000
snap 10b_note_gone
expect state loop2=705600

# SAMPLE EDIT shows the new length (the cuts are even ones, F2 dim)
tap sampleedit
wait 600
snap 11_edit

# F7 SAVE WAV (U8b): the slot as Samples/slot2_001.wav, the next one takes the next number; the job is the kit's, its strip the progress and the result
tap f7
snap 11a_saving
tap f6
snap 11b_refused_saving
wait 1500
snap 11c_saved
expect file Samples/slot2_001.wav
tap f7
wait 1500
expect file Samples/slot2_002.wav

# Leaving SAMPLE EDIT gives the save up: the other screens write the slot, so no half file stays
tap f7
tap sampleload
wait 400
expect nofile Samples/slot2_003.wav
tap sampleedit
wait 300

# F8 returns to the screen SAMPLE LOAD came from
tap sampleload
wait 400
tap f8
wait 200
snap 12_back

# a slot that records (the transport runs, the take starts at the bar line, and slot 1 is the one SAMPLE REC took) is refused
input tone 220 -8
tap samplerec
wait 200
tap play
wait 500
tap rec
wait 2500
snap 13a_recording
tap sampleload
wait 300
tap up
tap up
tap f1
wait 200
snap 13_busy
expect state screen=SAMPLE_LOAD
