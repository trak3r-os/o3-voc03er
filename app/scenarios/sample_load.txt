# SAMPLE LOAD (U8a): SAMPLE LOAD opens the kit's file picker on Apps/o3_voc03er/ (.wav only); UP and DOWN name the slot; F1 loads the file under the cursor into it;
# F8 returns to the screen it came from. A mono file, a stereo file over the same slot (REPLACES SLOT 1, F1 confirms: same length, so the slot is cleared first), a 48 kHz
# and an 8-bit file refused, any other key cancelling the ask, a damaged file refused, a 20 s stereo file clipped to 16 s (the read stops when the stage is full), a slot
# that records refused. The simulator shows the screen and the unit's length; it cannot say how it sounds. The card needs the WAVs of sample_load_wavs.py:
#   mkdir -p <card> <dir> && python3 -I app/scenarios/sample_load_wavs.py <card>/Apps/o3_voc03er
#   build-host/app/o3_voc03er_sim --scenario app/scenarios/sample_load.txt --card <card> --out <dir>
# Files sorted: a_mono, b_stereo (same length as a_mono), c_48k, d_8bit, e_long20, f_broken.
wait 300
expect state card=ready
tap sampleload
wait 600
snap 00_list
expect state loop1=0

# a_mono.wav (1.5 s) into the empty slot 1: no ask
tap f1
wait 150
snap 01_reading
wait 600
snap 02_loaded
expect state screen=SAMPLE_LOAD

# b_stereo.wav into the same slot: it holds a sample, so it asks; F1 confirms; the slot keeps the length
turn 1
tap f1
wait 100
snap 03_ask
tap f1
wait 1200
snap 04_replaced
expect state screen=SAMPLE_LOAD

# c_48k.wav: asks, F1 confirms, then the file is refused and the slot is untouched (the list starts at its top after every job)
turn 2

wait 800
