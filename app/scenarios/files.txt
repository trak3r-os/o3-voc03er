# The card, from the app (o3-sdk kit/app/include/o3sdk/app/files.h): the pattern the palette pads wrote is saved (PATTERN F8), the octave keeps
# itself, a reboot brings both back, and a missing card is said, not ignored. Needs the simulator's card, which is a directory:
#
#   mkdir <dir> <card> && <build dir>/app/o3_voc03er_sim --scenario app/scenarios/files.txt --card <card> --out <dir>
#
# The files are <card>/Apps/o3_voc03er/pattern.o3a (and .bak) and config.o3a. The scenario stops at the first `expect` that is wrong. Start from an
# empty card directory (tests/app_test.cpp gives each scenario a fresh one).
wait 100
tap pattern  # the pattern is the PATTERN screen's; F8 saves it
tap rec  # REC: the palette pads (the lower two rows) write
expect state card=ready
expect nofile Apps/o3_voc03er/pattern.o3a
expect state steps=0
snap 00_card_ready

# WRITE and SAVE. Three palette pads (REC on) write three steps; PATTERN's F8 saves them; the strip says SAVED.
pad 24 50
pad 27 50
pad 31 50
expect state steps=3
tap f8
wait 60
expect state files=done
expect file Apps/o3_voc03er/pattern.o3a
snap 01_saved

# The setting (PERFORM's octave) saves itself a moment after it changes (an octave down, SHIFT + F7 then the encoder): no key, no strip line.
tap performance
press shift
tap f7
release shift
turn -1
wait 60
expect state octave=-1
expect file Apps/o3_voc03er/config.o3a

# The second save keeps the first as the backup.
tap pattern
tap rec  # REC went off with the screen
pad 29 50
tap f8
wait 60
expect file Apps/o3_voc03er/pattern.o3a.bak
expect state steps=4
snap 02_second_save

# REBOOT: the runtime and the app are made again; the pattern and the octave come back from the card with no key pressed.
reboot
wait 200
expect state octave=-1
expect state steps=4
snap 04_after_reboot

# No card: a save is refused at once and says NO CARD; the card back and it goes through.
tap pattern
card out
wait 20
tap f8
wait 20
expect state files=failed
snap 05_no_card
card in
wait 100
tap f8
wait 60
expect state files=done
